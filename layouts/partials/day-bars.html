{{- /* A month's daily bar chart, drawn at build time: one slot per day of the month, the day's number under every seventh, each day's value as its bar's tooltip. Input: dict "days" (a month's days from data/stats.json), "key" (accounts, messages or media_bytes), "label" (what it counts, for the chart's name), "mib" (show bytes as MiB), "from" (the first hour counted, yyyy-mm-ddThh; earlier days weren't). A day not counted has no bar and "—" as its tooltip. No script: the tooltips are SVG titles. */ -}} {{- $days := .days -}} {{- $key := .key -}} {{- $mib := .mib | default false -}} {{- $fromDay := substr (.from | default "") 0 10 -}} {{- $w := 700.0 -}} {{- $h := 120.0 -}} {{- $max := 0.0 -}} {{- range $days }}{{ $raw := index . $key }}{{ if and (ne $raw nil) (ge .date $fromDay) }}{{ if gt (float $raw) $max }}{{ $max = float $raw }}{{ end }}{{ end }}{{ end -}} {{- $scale := cond (gt $max 0.0) $max 1.0 -}} {{- $step := div $w 31.0 -}} {{- $bar := math.Max 2.0 (sub $step 4.0) -}}
