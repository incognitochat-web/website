{{- /* An hourly bar chart, drawn at build time: one bar per hour, a light line at each midnight, the day's date under it, and each hour's value as the bar's tooltip. Input: dict "hours" (data/stats.json's hours), "key" (accounts, messages or media_bytes), "label" (what it counts, for the chart's name), "mib" (show bytes as MiB), "from" (the first hour counted, yyyy-mm-ddThh; earlier ones weren't). An hour not counted has no bar and "—" as its tooltip. No script: the tooltips are SVG titles. */ -}} {{- $hours := .hours -}} {{- $key := .key -}} {{- $mib := .mib | default false -}} {{- $from := .from | default "" -}} {{- $w := 700.0 -}} {{- $h := 120.0 -}} {{- $n := len $hours -}} {{- $max := 0.0 -}} {{- range $hours }}{{ $raw := index . $key }}{{ if and (ne $raw nil) (ge .t $from) }}{{ if gt (float $raw) $max }}{{ $max = float $raw }}{{ end }}{{ end }}{{ end -}} {{- $scale := cond (gt $max 0.0) $max 1.0 -}} {{- $step := div $w (cond (gt $n 0) (float $n) 1.0) -}} {{- $bar := math.Max 1.0 (sub $step 1.0) -}}
